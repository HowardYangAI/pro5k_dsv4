   348	# --- Page-split utilities: pbs=256 → pbs=64 ---
   349	# SGLang SWA KV cache footer layout per 256-token page:
   350	#   [data: 256 * 576 bytes] [scale: 256 * 8 bytes] [padding]
   351	# FlashInfer decode_dsv4 expects per 64-token page:
   352	#   [data: 64 * 576 bytes] [scale: 64 * 8 bytes] [padding to 37440]
   353	_PBS_SRC = 256  # SGLang physical page size
   354	_PBS_DST = 64  # FlashInfer page_block_size
   355	_NOPE_ROPE_STRIDE = 576  # bytes per token for nope+rope
   356	_SCALE_STRIDE = 8  # bytes per token for scale (7 + 1 pad)
   357	_BYTES_PER_DST_PAGE = (
   358	    _PBS_DST * _NOPE_ROPE_STRIDE + _PBS_DST * _SCALE_STRIDE
   359	)  # 64*576 + 64*8 = 37376 + 512 = 37888
   360	# Padded to 576 alignment
   361	
   362	_BYTES_PER_DST_PAGE_PADDED = math.ceil(_BYTES_PER_DST_PAGE / 576) * 576  # 37440
   363

   365	@triton.jit
   366	def _page_split_kernel(
   367	    src_ptr,
   368	    dst_ptr,
   369	    N_pages,
   370	    src_stride0: tl.constexpr,
   371	    dst_stride0: tl.constexpr,
   372	    DATA_PER_SUB: tl.constexpr,  # 64 * 576 = 36864
   373	    SCALE_PER_SUB: tl.constexpr,  # 64 * 8 = 512
   374	    SRC_SCALE_OFF: tl.constexpr,  # 256 * 576 = 147456
   375	    DST_SCALE_OFF: tl.constexpr,  # 64 * 576 = 36864
   376	    RATIO: tl.constexpr,  # 4
   377	    BLOCK_SIZE: tl.constexpr,
   378	    mask_ptr,
   379	    HAS_MASK: tl.constexpr,
   380	):
   381	    """Fused page-split: copy data+scale for all sub-pages in one kernel.
   382	
   383	    When HAS_MASK is set, only pages flagged in ``mask_ptr`` (int8, 1=touched)
   384	    are copied; untouched pages are skipped so the kernel no longer rewrites the
   385	    entire KV pool every decode step.
   386	    """
   387	    pid = tl.program_id(0)
   388	    page_idx = pid // RATIO
   389	    sub = pid % RATIO
   390	
   391	    if page_idx >= N_pages:
   392	        return
   393	
   394	    if HAS_MASK:
   395	        if tl.load(mask_ptr + page_idx) == 0:
   396	            return
   397	
   398	    # All layout strides/offsets are 8-byte aligned (asserted at the call
   399	    # site), so copy in u64 lanes instead of single bytes.
   400	    src_u64 = src_ptr.to(tl.pointer_type(tl.uint64))
   401	    dst_u64 = dst_ptr.to(tl.pointer_type(tl.uint64))
   402	    src_base = src_u64 + page_idx * (src_stride0 // 8)
   403	    dst_base = dst_u64 + (page_idx * RATIO + sub) * (dst_stride0 // 8)
   404	
   405	    DATA_U64: tl.constexpr = DATA_PER_SUB // 8
   406	    SCALE_U64: tl.constexpr = SCALE_PER_SUB // 8
   407	
   408	    # Copy data region: DATA_U64 u64 lanes from src offset sub*DATA_U64
   409	    data_src_off = sub * DATA_U64
   410	    for start in tl.range(0, DATA_U64, BLOCK_SIZE):
   411	        offs = start + tl.arange(0, BLOCK_SIZE)
   412	        mask = offs < DATA_U64
   413	        vals = tl.load(src_base + data_src_off + offs, mask=mask)
   414	        tl.store(dst_base + offs, vals, mask=mask)
   415	
   416	    # Copy scale region: SCALE_U64 u64 lanes
   417	    scale_src_off = SRC_SCALE_OFF // 8 + sub * SCALE_U64
   418	    for start in tl.range(0, SCALE_U64, BLOCK_SIZE):
   419	        offs = start + tl.arange(0, BLOCK_SIZE)
   420	        mask = offs < SCALE_U64
   421	        vals = tl.load(src_base + scale_src_off + offs, mask=mask)
   422	        tl.store(dst_base + DST_SCALE_OFF // 8 + offs, vals, mask=mask)
   423

   425	@triton.jit
   426	def _page_mark_kernel(
   427	    indices_ptr,
   428	    mask_ptr,
   429	    N_idx,
   430	    SRC_PBS: tl.constexpr,
   431	    BLOCK: tl.constexpr,
   432	):
   433	    """Mark touched source pages (1 byte each) from token-level indices.
   434	
   435	    ``indices`` are token indices into the pbs=SRC_PBS SWA pool; -1 = invalid.
   436	    Each valid token marks ``mask[token // SRC_PBS] = 1``. Concurrent stores of
   437	    the same value 1 are safe (no atomic needed).
   438	    """
   439	    pid = tl.program_id(0)
   440	    offs = pid * BLOCK + tl.arange(0, BLOCK)
   441	    valid = offs < N_idx
   442	    idx = tl.load(indices_ptr + offs, mask=valid, other=-1)
   443	    keep = valid & (idx >= 0)
   444	    page = idx // SRC_PBS
   445	    tl.store(mask_ptr + page, tl.full((BLOCK,), 1, tl.int8), mask=keep)
   446

   448	def _split_kv_pages_to_64(
   449	    kv_u8: torch.Tensor,
   450	    src_pbs: int,
   451	    touched_indices: Optional[torch.Tensor] = None,
   452	) -> torch.Tensor:
   453	    """Split pbs=N footer-format pages into pbs=64 footer-format pages.
   454	
   455	    When ``touched_indices`` (token-level int32 indices into the pbs=src_pbs
   456	    SWA pool, -1 = invalid) is provided, only the source pages that actually
   457	    contain a referenced token are copied. This avoids rewriting the entire KV
   458	    pool on every decode step (only ~2*batch pages are touched vs the full
   459	    pool). The output buffer is persistent and reused across steps; untouched
   460	    dst pages simply retain their (unreferenced) stale data.
   461	    """
   462	    assert src_pbs % _PBS_DST == 0 and src_pbs >= _PBS_DST
   463	    if src_pbs == _PBS_DST:
   464	        return kv_u8
   465	
   466	    N = kv_u8.shape[0]
   467	    ratio = src_pbs // _PBS_DST
   468	    num_dst_pages = N * ratio
   469	
   470	    from sglang.srt.runtime_context import get_resources
   471	
   472	    # Pre-allocated grow-only buffer for page-split output per device.
   473	    dev = kv_u8.device
   474	    buffers = get_resources().buffers
   475	    key = f"flash_mla_sm120_split:{dev}"
   476	    buf = buffers.get(key)
   477	    if buf is None or buf.shape[0] < num_dst_pages:
   478	        # The first allocation can happen under inference mode (autotune), but
   479	        # the buffer is written again during CUDA graph capture outside
   480	        # inference mode, where an inference tensor cannot be mutated.
   481	        with torch.inference_mode(False):
   482	            buf = torch.empty(
   483	                num_dst_pages,
   484	                _BYTES_PER_DST_PAGE_PADDED,
   485	                dtype=torch.uint8,
   486	                device=dev,
   487	            )
   488	        buffers[key] = buf
   489	    out = buf[:num_dst_pages]
   490	
   491	    # Get raw 2D view of source
   492	    src_2d = kv_u8
   493	    if src_2d.ndim == 4:
   494	        src_stride0 = src_2d.stride(0)
   495	        src_2d = torch.as_strided(src_2d, (N, src_stride0), (src_stride0, 1))
   496	    else:
   497	        src_stride0 = src_2d.stride(0)
   498	
   499	    use_mask = touched_indices is not None and touched_indices.numel() > 0
   500	    mask_ptr = src_2d  # dummy, never dereferenced when HAS_MASK is False
   501	    if use_mask:
   502	        # Persistent per-device int8 mask, zeroed each call (cheap memset,
   503	        # captured cleanly by CUDA graph). 1 = page is referenced this step.
   504	        mkey = f"flash_mla_sm120_mask:{dev}"
   505	        mbuf = buffers.get(mkey)
   506	        if mbuf is None or mbuf.shape[0] < N:
   507	            # The first allocation can happen under inference mode (autotune),
   508	            # but the buffer is zeroed again later during CUDA graph capture
   509	            # outside inference mode -- an inference tensor cannot be mutated
   510	            # there, so force a normal tensor.
   511	            with torch.inference_mode(False):
   512	                mbuf = torch.empty(N, dtype=torch.int8, device=dev)
   513	            buffers[mkey] = mbuf
   514	        mask = mbuf[:N]
   515	        mask.zero_()
   516	        idx_flat = touched_indices.reshape(-1).contiguous()
   517	        if idx_flat.dtype != torch.int32:
   518	            idx_flat = idx_flat.to(torch.int32)
   519	        _MARK_BLOCK = 1024
   520	        _page_mark_kernel[(triton.cdiv(idx_flat.numel(), _MARK_BLOCK),)](
   521	            idx_flat,
   522	            mask,
   523	            idx_flat.numel(),
   524	            src_pbs,  # SRC_PBS
   525	            _MARK_BLOCK,
   526	        )
   527	        mask_ptr = mask
   528	
   529	    assert src_stride0 % 8 == 0 and _BYTES_PER_DST_PAGE_PADDED % 8 == 0
   530	    grid = (N * ratio,)
   531	    _page_split_kernel[grid](
   532	        src_2d,
   533	        out,
   534	        N,
   535	        src_stride0,
   536	        _BYTES_PER_DST_PAGE_PADDED,
   537	        _PBS_DST * _NOPE_ROPE_STRIDE,  # DATA_PER_SUB = 36864
   538	        _PBS_DST * _SCALE_STRIDE,  # SCALE_PER_SUB = 512
   539	        src_pbs * _NOPE_ROPE_STRIDE,  # SRC_SCALE_OFF = 147456
   540	        _PBS_DST * _NOPE_ROPE_STRIDE,  # DST_SCALE_OFF = 36864
   541	        ratio,  # RATIO = 4
   542	        1024,  # BLOCK_SIZE
   543	        mask_ptr,
   544	        use_mask,  # HAS_MASK
   545	    )
   546	
   547	    bpt = _NOPE_ROPE_STRIDE + _SCALE_STRIDE  # 584
   548	    return out.as_strided(
   549	        (num_dst_pages, _PBS_DST, 1, bpt),
   550	        (_BYTES_PER_DST_PAGE_PADDED, bpt, bpt, 1),
   551	    )

   554	def _flash_mla_flashinfer(
   555	    q,
   556	    k_cache,
   557	    indices,
   558	    topk_length,
   559	    attn_sink,
   560	    head_dim_v,
   561	    softmax_scale,
   562	    extra_k_cache,
   563	    extra_indices,
   564	    extra_topk_length,
   565	):
   566	    """FlashInfer SM120 sparse MLA via the paged-attention dispatcher.
   567	
   568	    SGLang SWA pool uses page_size=256 (footer format: 256*576 bytes data + 256*8 bytes scale).
   569	    FlashInfer decode_dsv4 fast path requires page_block_size=64 (footer: 64*576 + 64*8).
   570	    We split 256-token pages into 4 virtual 64-token pages.
   571	    Token indices are invariant under page-split (identity mapping).
   572	    """
   573	    from flashinfer.mla._sparse_mla_sm120 import (
   574	        _DECODE_MAX_TOKENS as _FI_DECODE_MAX_TOKENS,
   575	    )
   576	    from flashinfer.mla._sparse_mla_sm120 import (
   577	        _sparse_mla_sm120_paged_attention,
   578	    )
   579	
   580	    B, _, H, D = q.shape  # (batch, 1, num_heads, head_dim)
   581	    dev = q.device
   582	
   583	    # Indices: no remapping needed (page-split preserves token addressing).
   584	    idx = indices.squeeze(1) if indices.dim() == 3 else indices
   585	
   586	    # --- Page-split: convert pbs=N kv_cache to pbs=64 view ---
   587	    # Only the SWA pages actually referenced by `idx` are copied (the rest of
   588	    # the persistent dst buffer is left untouched and never read).
   589	    kv_u8 = k_cache.view(torch.uint8) if k_cache.dtype != torch.uint8 else k_cache
   590	    src_pbs = k_cache.shape[1] if k_cache.ndim >= 3 else _PBS_SRC
   591	    kv_64 = (
   592	        _split_kv_pages_to_64(kv_u8, src_pbs, touched_indices=idx)
   593	        if src_pbs != _PBS_DST
   594	        else kv_u8
   595	    )
   596	
   597	    extra_kv_u8 = (
   598	        extra_k_cache.view(torch.uint8)
   599	        if extra_k_cache is not None and extra_k_cache.dtype != torch.uint8
   600	        else extra_k_cache
   601	    )
   602	    extra_kv_64 = extra_kv_u8
   603	
   604	    extra_idx = (
   605	        extra_indices.squeeze(1)
   606	        if extra_indices is not None and extra_indices.dim() == 3
   607	        else extra_indices
   608	    )
   609	
   610	    output = torch.empty(B, H, head_dim_v, dtype=torch.bfloat16, device=dev)
   611	    out_lse = torch.empty(B, H, dtype=torch.float32, device=dev)
   612	
   613	    # Use split-K for decode-sized batches and paged attention otherwise.
   614	    if B <= _FI_DECODE_MAX_TOKENS:
   615	        topk = idx.shape[-1]
   616	        extra_topk = extra_idx.shape[-1] if extra_idx is not None else 0
   617	        _BI = 64
   618	        num_splits = (topk + _BI - 1) // _BI + (
   619	            (extra_topk + _BI - 1) // _BI if extra_topk > 0 else 0
   620	        )
